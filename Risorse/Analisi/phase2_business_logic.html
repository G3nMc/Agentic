2.2 Register Route for Order Creation

from flask import Blueprint, request, g from domain.services.order_management_service import ( OrderManagementService, CreateOrderRequest ) from domain.shared.task_result import TaskResult import logging logger = logging.getLogger(__name__) order_management_routes = Blueprint('order_management_routes', __name__) @order_management_routes.route("/create_with_lines", methods=["POST"]) def create_order_with_lines(): try: data = request.json # Parse request order_request = CreateOrderRequest( customer_id=data.get('customer_id'), order_date=data.get('order_date'), lines=data.get('lines', []), shipping_address_id=data.get('shipping_address_id') ) # Get current user from JWT user_id = g.get('current_user_id', 1) # Create service and execute service = OrderManagementService() service.order_service = g.order_service service.order_line_service = g.order_line_service service.product_service = g.product_service service.customer_service = g.customer_service result = service.create_order_with_lines(order_request, user_id) return result.to_response() except Exception as e: logger.error(f"Error in create_order_with_lines: {e}", exc_info=True) return TaskResult.failure_response( message="Internal server error" ).to_response()
