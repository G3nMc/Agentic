class ConversationHistory:
    """Conversation history with separated system prompts and clean turns."""

    def __init__(self):
        # System prompts keyed by name. Each key maps to one system instruction.
        # Setting the same key replaces the content (no duplication).
        self._system_prompts: Dict[str, str] = {}

        # Clean conversation turns: only real user/assistant exchanges
        # + tool results (which are part of the working conversation).
        self._turns: List[Dict[str, str]] = []

    # ── System prompts (keyed, static) ──
    def set_system_prompt(self, key: str, content: str) -> None:
        """Set or replace a system prompt by key."""
        self._system_prompts[key] = content

    def remove_system_prompt(self, key: str) -> None:
        self._system_prompts.pop(key, None)

    def get_system_prompt(self, key: str) -> Optional[str]:
        return self._system_prompts.get(key)

    def clear_system_prompts(self) -> None:
        self._system_prompts.clear()

    # ── Clean turns (user/assistant only) ──
    def add_user(self, content: str) -> None:
        self._turns.append({"role": "user", "content": content})

    def add_assistant(self, content: str) -> None:
        self._turns.append({"role": "assistant", "content": content})

    def add_turn(self, role: str, content: str) -> None:
        self._turns.append({"role": role, "content": content})

    def pop(self) -> Optional[Dict[str, str]]:
        """Remove and return the last turn, or None if empty."""
        return self._turns.pop() if self._turns else None

    def last_turn(self) -> Optional[Dict[str, str]]:
        return self._turns[-1] if self._turns else None

    # ── Build the message list for the model ──
    def to_messages(self) -> List[Dict[str, str]]:
        """Build the full message list: system prompts first, then turns.

        System prompts are emitted in insertion order (Python 3.7+ dicts
        preserve order), so the base system prompt is always first.
        """
        system_msgs = [
            {"role": "system", "content": content}
            for content in self._system_prompts.values()
        ]
        return system_msgs + self._turns

    # ── Utility ──
    def reset_turns(self) -> None:
        """Clear turns but keep system prompts."""
        self._turns = []

    def reset_all(self) -> None:
        """Clear everything."""
        self._system_prompts.clear()
        self._turns = []

    def import_external_history(self, external: List[Dict[str, Any]]) -> None:
        """Append caller-supplied user/assistant turns."""
        for msg in external:
            role = str(msg.get("role") or "").strip().lower()
            content = str(msg.get("content") or "")
            if role in ("user", "assistant") and content.strip():
                self._turns.append({"role": role, "content": content})

    @property
    def turns(self) -> List[Dict[str, str]]:
        return self._turns

    @property
    def system_prompts(self) -> Dict[str, str]:
        return self._system_prompts

⚠️ Sfide e rischi
1. Tool results nei turns
I tool results ("Tool read_file returned: ...") devono restare nei turni perché il modello li referenzia. Ma il contenuto include [INTERNAL: Continue...] che è un'istruzione di sistema. Soluzione: separare l'istruzione di sistema (chiave "tool_followup") dal risultato del tool (nei turns).

2. Feedback efimeri
I feedback come schema_feedback, repeat_warning, ecc. devono essere efimeri: impostati per la prossima iterazione e poi rimossi o sostituiti. Con il sistema a chiavi, basta chiamare set_system_prompt("schema_feedback", nuovo_contenuto) che sostituisce il vecchio.

3. Ordine dei system prompts
Alcuni provider richiedono che i messaggi system siano all'inizio. OpenAI e Ollama lo accettano. Anthropic accetta solo un system message. Bisogna verificare per ogni backend.

4. 214 riferimenti da aggiornare
conversation_history è referenziato 214 volte in run_loop.py. Ogni riferimento deve essere aggiornato per usare la nuova API (add_user, add_assistant, to_messages, ecc.).

5. Compatibilità con orchestrator_multi.py
L'executor stage in orchestrator_multi.py accede a executor_orch.conversation_history (linea 528) per iniettare il previous execution context. Bisogna aggiornare anche quello.

6. Trimming
Le funzioni trim_history e trim_history_by_tokens in history.py operano su una lista piatta. Bisogna aggiornarle per operare sui turni separati dal system.
